CREATE PROCEDURE [dbo].[sp_InsertInward]
    @CompanyId INT,
    @Colour NVARCHAR(100),
    @DesignName NVARCHAR(150),
    @StyleNo NVARCHAR(100),
    @InwardDcNo NVARCHAR(100),
    @PoNo NVARCHAR(100) = NULL,
    @UploadURL NVARCHAR(500),
    @CreatedBy INT
AS
BEGIN
    SET NOCOUNT ON;
    -- Trim strings
    SET @Colour = LTRIM(RTRIM(@Colour));
    SET @DesignName = LTRIM(RTRIM(@DesignName));
    SET @StyleNo = LTRIM(RTRIM(@StyleNo));
    SET @InwardDcNo = LTRIM(RTRIM(@InwardDcNo));
    IF @PoNo IS NOT NULL
        SET @PoNo = LTRIM(RTRIM(@PoNo));

    INSERT INTO Inward (
        CompanyId, 
        Colour, 
        DesignName, 
        StyleNo, 
        InwardDcNo,
        PoNo,
        UploadURL, 
        CreatedBy, 
        CreatedDate
    )
    VALUES (
        @CompanyId, 
        @Colour, 
        @DesignName, 
        @StyleNo, 
        @InwardDcNo,
        @PoNo,
        @UploadURL, 
        @CreatedBy, 
        GETDATE()
    );

    SELECT SCOPE_IDENTITY() AS InwardId;
END
